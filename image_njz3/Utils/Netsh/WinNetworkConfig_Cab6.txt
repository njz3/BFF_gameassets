# Static IPv4 configuration for RaceCab6
netsh interface ipv4 set address name="!NETIF!" static 192.168.1.9 255.255.255.0 192.168.1.254
netsh interface ipv4 set dnsservers name="!NETIF!" static 8.8.8.8 primary validate=no
netsh interface ipv4 add dnsservers name="!NETIF!" 8.8.4.4 index=2 validate=no
